%%  Creates a hyperlink with a custom underline colour. The colour of the link text itself is

%%  \resumeHeading{text}
%%  ------------------------------------------------------------------------------------------------
%%  Creates a non-numbered `\section` heading.

\newcommand{\resumeHeading}[1]{%

%%  \resumeSubheading[vspace]{company}{role}{employment-type}{dates}
%%  ------------------------------------------------------------------------------------------------
%%  Creates a custom sub-section heading. Used for job role headings.
%%
%%  Arguments:
%%  - `vspace`: Space above the subheading - optional, default = 12pt.
%%  - `company`: The name of the company or organisation.
%%  - `role`: The job title or role.
%%  - `employment-type`: The type of employment (eg., contract, permanent). If empty, the
%%    parentheses are omitted.
%%  - `dates`: The dates of employment (eg., 2023--present).
%%
%%  Example:
%%    \resumeSubheading{ACME Corp}{Software Engineer}{contract}{2023--present}
%%
\newcommand{\resumeSubheading}[5][12pt]{%
  \vspace{#1}
  {\scshape{\textls[\trackingBody]{#2}}}\quad\small{\textls[\trackingBody]{#3}}\if\relax\detokenize{#4}\relax\else\ \small{(\textls[\trackingBody]{#4})}\fi\hspace{0.5em}{\color{quiet}·}\hspace{0.5em}\small{\textls[\trackingBody]{#5}}
  \vspace{6pt}%
}
